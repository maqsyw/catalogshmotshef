function openPostAndClose(url, element) { if (!url || url === '#') return; // Визуальный отклик на кнопке: показываем пользователю, что клик сработал if (element) { element.innerText = "Переход..."; element.style.opacity = "0.7"; } if (window.Telegram && window.Telegram.WebApp) { // Запускаем открытие ссылки window.Telegram.WebApp.openTelegramLink(url); } else { window.open(url, '_blank'); } }
